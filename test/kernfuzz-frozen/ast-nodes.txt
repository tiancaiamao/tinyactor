# ast-nodes.txt — frozen AST node-shape table (DELIV-1)
#
# Authority: test/compiler/parser-ast.ta (header comment + cases() rows).
# This file is the frozen "完整节点形态清单" required by task-snapshots §2:
#   - one line per structural node kind = the list-node car symbol
#   - shape note derived from the parser-ast.ta grammar rows
#   - verified against the actual corpus snapshots (see §Diff below)
#
# Rendering conventions (canonical s-expr, from parser-ast.ta render()):
#   int -> 42           string -> "hi" (quoted, escapes decoded)
#   symbol -> name      nil -> nil     true/false -> true/false
#   list -> (a b c)     dotted pair -> (a . b)
#
# A list whose car is a symbol is a call or a structural node; a list whose
# car is a pair is a chained call ((f x) y).  Only the structural kinds are
# tabulated below; every other symbol-headed list is an ordinary function
# call whose head is the (possibly dotted, e.g. mod.fn) callee name.

## Structural node kinds (car symbol -> shape)

define        (define (name params...) body)        ; fn definition
              ; anonymous fn at top level: (define (nil) x)
              ; empty body: (define (f) nil)
              ; multi-form body: (define (f) (begin a b))
define_pub    (define_pub (name params...) body)    ; pub fn (same shape as define)
type-sig      (type-sig name (param-types...) ret-type)
              ; ret-type = nil when absent; param types may nest:
              ;   compound: (List int)      function: (arrow a b)
              ;   quoted var: (quote a)
type          (type Name (params...) (quote Variant) (Ctor args...)...)
              ; params = nil for non-parametric; nullary variants are (quote A)
const         (const NAME expr)                     ; compile-time constant
              ; NAME reference inside expr is quoted: (quote COUNT)
import        (import "mod")                        ; module name is a string
external_fn   (external_fn mod.name (param-types...) ret-type)
              ; ret-type = nil when absent
begin         (begin form...)                       ; block / multi-form body
let           (let x val body)                      ; flat bind, scopes over rest
              ; trailing let: (let x 1 nil); non-leading let is unthreaded:
              ;   (begin 1 (let x 2) x)
if            (if cond then else)                   ; else = nil when absent
lambda        (lambda (params...) body ret-type)    ; anonymous fn
              ; untyped: ret-type = nil; no-parens: (lambda nil 42 nil)
match         (match scrutinee (pat [guard] body)...)
              ; arm = (pat body) or (pat guard body); empty: (match x)
receive       (receive (pat [guard] body)...)       ; same arm shape as match
quote         (quote sym)                           ; quoted symbol: nullary ctor
              ;   ref, ctor pattern, quoted type var
cons          (cons a b)                            ; list literal / ctor-call desugar
              ; ctor call: (cons (quote Name) args...); [] desugars to nil
float         (float "3.14")                        ; float literal (string payload)
arrow         (arrow a b)                           ; function type (right assoc)
bind          (bind expr (lambda (x) rest))         ; `use x <- expr; rest` desugar
and           (and a b)                             ; && right assoc
or            (or a b)                              ; || left assoc
=             (= a b)                               ; == (renamed), left assoc
!=            (!= a b)
<             (< a b)
>             (> a b)
<=            (<= a b)
>=            (>= a b)
+             (+ a b)
-             (- a b)
*             (* a b)
/             (/ a b)
%             (% a b)

## Atoms (leaf values, not list nodes)
int          42 / -42
string       "hi"          ; quoted, escapes decoded
symbol       name / foo.bar
nil          nil
true         true
false        false

## Call forms (any symbol-headed list not in the table above)
call         (f a b)               ; callee head = any symbol, incl. dotted mod.fn
chained      ((f x) y)             ; pair head = chained call f(x)(y)
ctor-call    (cons (quote Name) a b ...)   ; Name(...) desugars to cons+quote
